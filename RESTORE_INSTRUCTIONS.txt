import requests, pandas as pd
print("Downloading master contract (190,304 instruments)...")

# Full contract (NSE EQ stocks + NFO + BSE)
r = requests.get(
    "https://margincalculator.angelbroking.com/OpenAPI_File/files/OpenAPIScripMaster.json",
    timeout=30)
df = pd.DataFrame(r.json())
df.to_csv("MasterContract_ALL.csv", index=False)
print(f"✅ MasterContract_ALL.csv: {len(df):,} instruments")

# NFO only (for options/futures trading)
nfo = df[df["exch_seg"].str.upper()=="NFO"] if "exch_seg" in df.columns else df
nfo.to_csv("MasterContract_NFO.csv", index=False)
print(f"✅ MasterContract_NFO.csv: {len(nfo):,} NFO instruments")

# NSE EQ stocks (for historical data token lookup)
nse = df[df["exch_seg"].str.upper()=="NSE"] if "exch_seg" in df.columns else df
print(f"✅ NSE EQ stocks: {len(nse):,}")

# Spot-check key tokens
for sym, expected in [("RELIANCE-EQ","2885"),("HDFCBANK-EQ","1333")]:
    row = df[df["symbol"]==sym]
    tok = str(row.iloc[0]["token"]) if len(row) else "NOT FOUND"
    status = "✅" if tok == expected else "⚠️"
    print(f"  {status} {sym}: token={tok}")
PYEOF
